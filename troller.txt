giveToolBtn.MouseButton1Click:Connect(function()
	local backpack = player:FindFirstChildOfClass("Backpack") or player:WaitForChild("Backpack")
	local character = player.Character
	
	if not backpack:FindFirstChild("SelectPlayerTool") and not (character and character:FindFirstChild("SelectPlayerTool")) then
		local tool = Instance.new("Tool")
		tool.Name = "SelectPlayerTool"
		tool.RequiresHandle = false
		tool.ToolTip = "Clicca su un player per selezionarlo!"
		
		tool.Activated:Connect(function()
			local mouse = player:GetMouse()
			local target = mouse.Target
			if target and target.Parent then
				local charModel = target.Parent
				local foundPlayer = Players:GetPlayerFromCharacter(charModel)
				if not foundPlayer then
					charModel = charModel.Parent
					foundPlayer = Players:GetPlayerFromCharacter(charModel)
				end
				
				if foundPlayer and foundPlayer ~= player then
					selectedTarget = foundPlayer
					giveToolBtn.Text = "✅ Target Selezionato: " .. foundPlayer.Name
					task.wait(1.5)
					giveToolBtn.Text = "🛠️ Give Select Tool (Prendi Tool per selezionare)"
				end
			end
		end)
		
		tool.Parent = backpack
		giveToolBtn.Text = "✨ Tool aggiunto nel tuo inventario!"
		task.wait(1.5)
		giveToolBtn.Text = "🛠️ Give Select Tool (Prendi Tool per selezionare)"
	else
		giveToolBtn.Text = "⚠️ Hai già il tool nell'inventario!"
		task.wait(1.5)
		giveToolBtn.Text = "🛠️ Give Select Tool (Prendi Tool per selezionare)"
	end
end)

local function createTroolAction(text, callback)
	local btn = Instance.new("TextButton")
	btn.Size = UDim2.new(1, 0, 0, 32)
	btn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
	btn.BorderSizePixel = 0
	btn.Font = Enum.Font.GothamMedium
	btn.Text = text
	btn.TextColor3 = Color3.fromRGB(240, 240, 240)
	btn.TextSize = 13
	btn.Parent = troolSection

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 8)
	corner.Parent = btn

	local stroke = Instance.new("UIStroke")
	stroke.Color = Color3.fromRGB(80, 80, 80)
	stroke.Transparency = 0.6
	stroke.Parent = btn

	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(50, 50, 50)}):Play()
	end)

	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(30, 30, 30)}):Play()
	end)

	btn.MouseButton1Click:Connect(function()
		if selectedTarget and selectedTarget.Parent then
			callback(btn)
		else
			local oldText = btn.Text
			btn.Text = "⚠️ Seleziona prima un player!"
			TweenService:Create(btn, TweenInfo.new(0.1), {TextColor3 = Color3.fromRGB(255, 100, 100)}):Play()
			task.wait(1.5)
			TweenService:Create(btn, TweenInfo.new(0.1), {TextColor3 = Color3.fromRGB(240, 240, 240)}):Play()
			btn.Text = oldText
		end
	end)
end

createTroolAction("📍 Teleport to Player", function()
	local char = player.Character
	local targetChar = selectedTarget.Character
	if char and targetChar then
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")

createTroolAction("🌪️ Fling Player (ON / OFF)", function(btn)
	flingActive = not flingActive
	if flingActive then
		btn.Text = "🌪️ Fling Player [ON]"
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(240, 240, 240), TextColor3 = Color3.fromRGB(15, 15, 15)}):Play()
		
		task.spawn(function()
			while flingActive and selectedTarget and selectedTarget.Character do
				local myChar = player.Character
				local targetChar = selectedTarget.Character
				if myChar and targetChar then
					local myHRP = myChar:FindFirstChild("HumanoidRootPart")
					local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
					local myHum = myChar:FindFirstChildOfClass("Humanoid")
					
					if myHRP and targetHRP and myHum then
						myHRP.CFrame = targetHRP.CFrame
						myHRP.AssemblyLinearVelocity = Vector3.new(35000, 35000, 35000)
						myHRP.AssemblyAngularVelocity = Vector3.new(50000, 50000, 50000)
					end
				end
				RunService.Heartbeat:Wait()
			end
		end)
	else
		flingActive = false
		btn.Text = "🌪️ Fling Player (ON / OFF)"
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(30, 30, 30), TextColor3 = Color3.fromRGB(240, 240, 240)}):Play()
		local myHRP = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if myHRP then
			myHRP.AssemblyLinearVelocity = Vector3.zero
			myHRP.AssemblyAngularVelocity = Vector3.zero
		end
	end
end)

createTroolAction("🏃 Follow Player (ON / OFF)", function(btn)
	if not followConnection then
		followConnection = RunService.RenderStepped:Connect(function()
			if selectedTarget and selectedTarget.Character then
				local char = player.Character
				local targetChar = selectedTarget.Character
				if char and targetChar then
					local humanoid = char:FindFirstChildOfClass("Humanoid")
					local hrp = char:FindFirstChild("HumanoidRootPart")
					local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
					
					if humanoid and hrp and targetHRP then
						local distance = (hrp.Position - targetHRP.Position).Magnitude
						if distance > 35 then
							hrp.CFrame = targetHRP.CFrame * CFrame.new(0, 0, 3)
						else
							humanoid:MoveTo(targetHRP.Position)
						end
					end
				end
			end
		end)
		btn.Text = "🏃 Follow Player [ON]"
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(240, 240, 240), TextColor3 = Color3.fromRGB(15, 15, 15)}):Play()
	else
		if followConnection then followConnection:Disconnect() followConnection = nil end
		local humanoid = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
		if humanoid then humanoid:MoveTo(player.Character.HumanoidRootPart.Position) end
		btn.Text = "🏃 Follow Player (ON / OFF)"
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(30, 30, 30), TextColor3 = Color3.fromRGB(240, 240, 240)}):Play()
	end
end)

local function refreshPlayerList()
	-- Conserva la posizione dello scroll durante l'aggiornamento della lista.
	local savedCanvasPosition = playerListContainer.CanvasPosition

	if selectedTarget and not selectedTarget.Parent then
		selectedTarget = nil
		if isViewing and player.Character and player.Character:FindFirstChild("Humanoid") then
			workspace.CurrentCamera.CameraSubject = player.Character.Humanoid
			isViewing = false
		end
		if followConnection then followConnection:Disconnect() followConnection = nil end
		if backAttachConnection then backAttachConnection:Disconnect() backAttachConnection = nil end
		if kickHeadConnection then kickHeadConnection:Disconnect() kickHeadConnection = nil end
		flingActive = false
	end

	for _, child in ipairs(playerListContainer:GetChildren()) do
		if child:IsA("TextButton") then
			child:Destroy()
		end
	end

	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= player then
			local pBtn = Instance.new("TextButton")
			pBtn.Size = UDim2.new(1, 0, 0, 54)
			if selectedTarget == p then
				pBtn.BackgroundColor3 = Color3.fromRGB(240, 240, 240)
			else
				pBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
			end
			pBtn.BorderSizePixel = 0
			pBtn.Text = ""
			pBtn.Parent = playerListContainer

			local pCorner = Instance.new("UICorner")
			pCorner.CornerRadius = UDim.new(0, 8)
			pCorner.Parent = pBtn

			local avatarImg = Instance.new("ImageLabel")
			avatarImg.Size = UDim2.new(0, 44, 0, 44)
			avatarImg.Position = UDim2.new(0, 5, 0.5, -22)
			avatarImg.BackgroundTransparency = 1
			avatarImg.ScaleType = Enum.ScaleType.Fit

			pcall(function()
				local image, isReady = Players:GetUserThumbnailAsync(
					p.UserId,
					Enum.ThumbnailType.AvatarThumbnail,
					Enum.ThumbnailSize.Size420x420
				)
				avatarImg.Image = image
			end)
			avatarImg.Parent = pBtn

			local imgCorner = Instance.new("UICorner")
			imgCorner.CornerRadius = UDim.new(0, 6)
			imgCorner.Parent = avatarImg

			local nameLabel = Instance.new("TextLabel")
			nameLabel.Size = UDim2.new(1, -60, 1, 0)
			nameLabel.Position = UDim2.new(0, 56, 0, 0)
			nameLabel.BackgroundTransparency = 1
			nameLabel.Font = Enum.Font.GothamMedium
			nameLabel.Text = p.Name
			if selectedTarget == p then

local function sendChatMessage(message)
	local sent = false
	if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
		local textChannels = TextChatService:FindFirstChild("TextChannels")
		local generalChannel = textChannels and textChannels:FindFirstChild("RBXGeneral")
		if generalChannel then
			pcall(function()
				generalChannel:SendAsync(message)
				sent = true
			end)
		end
	end

	if not sent then
		local defaultChat = ReplicatedStorage:FindFirstChild("DefaultChatSystemChatEvents")
		local sayMessage = defaultChat and defaultChat:FindFirstChild("SayMessageRequest")
		if sayMessage then
			pcall(function()
				sayMessage:FireServer(message, "All")
			end)
		end
	end
end

local function startSnake(character)
	if not character then return end
	local humanoid = character:WaitForChild("Humanoid")
	local rootPart = character:WaitForChild("HumanoidRootPart")

	local swimSound = rootPart:FindFirstChild("RemawedSnakeSwimSound")
	if not swimSound then
		swimSound = Instance.new("Sound")
		swimSound.Name = "RemawedSnakeSwimSound"
		swimSound.SoundId = "rbxasset://sounds/action_swim.mp3"
		swimSound.Volume = 0.85
		swimSound.PlaybackSpeed = 1.15
		swimSound.Parent = rootPart
	end

	humanoid.AutoRotate = false
	lastSwimSoundTime = 0

	snakeConnection = RunService.Heartbeat:Connect(function()
		if not snakeActive or not character.Parent then return end

		humanoid:SetStateEnabled(Enum.HumanoidStateType.Swimming, true)
		humanoid:ChangeState(Enum.HumanoidStateType.Swimming)
		humanoid.Jump = true

		local moveDir = humanoid.MoveDirection
		local currentPos = rootPart.Position

		local lookDir = moveDir.Magnitude > 0.05 and moveDir or Vector3.new(rootPart.CFrame.LookVector.X, 0, rootPart.CFrame.LookVector.Z)
		if lookDir.Magnitude < 0.01 then
			lookDir = Vector3.new(0, 0, -1)
		else
			lookDir = lookDir.Unit
		end

		local flatCFrame = CFrame.lookAt(currentPos, currentPos + lookDir) * CFrame.Angles(math.rad(-90), 0, 0)
		rootPart.CFrame = flatCFrame
		rootPart.AssemblyAngularVelocity = Vector3.zero

		if moveDir.Magnitude > 0.05 then
			rootPart.AssemblyLinearVelocity = Vector3.new(
				moveDir.X * 40,
				-35,
				moveDir.Z * 40
			)
		else
			rootPart.AssemblyLinearVelocity = Vector3.new(0, -35, 0)
		end

		if tick() - lastSwimSoundTime >= 0.42 then
			lastSwimSoundTime = tick()
			swimSound:Play()
		end
	end)
end

	if character then

local function startBug(character)
	if not character then return end
	local humanoid = character:WaitForChild("Humanoid")
	local rootPart = character:WaitForChild("HumanoidRootPart")

	savedBugCFrame = rootPart.CFrame
	targetIndex = 1

	bugConnection = RunService.Heartbeat:Connect(function()
		if not bugActive or not character.Parent then return end

		humanoid.PlatformStand = true

		local targets = {}
		for _, otherPlayer in ipairs(Players:GetPlayers()) do
			if otherPlayer ~= player and otherPlayer.Character then
				local targetRoot = otherPlayer.Character:FindFirstChild("HumanoidRootPart")
				local targetHum = otherPlayer.Character:FindFirstChildOfClass("Humanoid")
				if targetRoot and targetHum and targetHum.Health > 0 then
					table.insert(targets, targetRoot)
				end
			end
		end

		local destinationPos
		if #targets > 0 then
			if targetIndex > #targets then
				targetIndex = 1
			end
			local currentTarget = targets[targetIndex]
			local offset = Vector3.new(
				math.random(-2, 2),
				math.random(-1, 2),
				math.random(-2, 2)
			)
			destinationPos = currentTarget.Position + offset
			targetIndex += 1
		else
			local basePos = savedBugCFrame and savedBugCFrame.Position or rootPart.Position
			destinationPos = basePos + Vector3.new(
				math.random(-35, 35),
				math.random(2, 25),
				math.random(-35, 35)
			)
		end

		local randomAngles = CFrame.Angles(
			math.rad(math.random(0, 360)),
			math.rad(math.random(0, 360)),
			math.rad(math.random(0, 360))
		)
		rootPart.CFrame = CFrame.new(destinationPos) * randomAngles

		local dir = (destinationPos - rootPart.Position)
		if dir.Magnitude < 0.01 then
			dir = Vector3.new(math.random(-10, 10), math.random(-10, 10), math.random(-10, 10))
		end
		dir = dir.Unit

		local ultraSpeed = math.random(32000, 45000)
		rootPart.AssemblyLinearVelocity = dir * ultraSpeed
		rootPart.AssemblyAngularVelocity = Vector3.new(
			math.random(-35000, 35000),
			math.random(-35000, 35000),
			math.random(-35000, 35000)
		)
	end)
end

local dragging = false
local dragInput, mousePos, framePos

topBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		mousePos = input.Position
		framePos = mainFrame.Position
		
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - mousePos
		mainFrame.Position = UDim2.new(
			framePos.X.Scale,
			framePos.X.Offset + delta.X,
			framePos.Y.Scale,
			framePos.Y.Offset + delta.Y
		)
	end
end)
