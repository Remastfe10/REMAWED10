-- Funzione per trovare la Part "Teleport1" nel workspace
local function getTeleport1Part()
	local p = workspace:FindFirstChild("Teleport1", true)
	if p and p:IsA("BasePart") then
		return p
	end
	return nil
end

-- Simula i click
local function simulateClicks(count)
	local viewportSize = workspace.CurrentCamera.ViewportSize
	local centerX = viewportSize.X / 2
	local centerY = viewportSize.Y / 2
	
	for i = 1, count do
		VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, true, game, 0)
		task.wait(0.03)
		VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, false, game, 0)
		task.wait(0.05)
	end
end

-- Target più vicino (Auto Slap)
local function getClosestTarget()
	local myChar = LocalPlayer.Character
	if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end

	local myPos = myChar.HumanoidRootPart.Position
	local closestTarget = nil
	local closestDist = DISTANZA_MASSIMA

	for _, player in pairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			local targetRoot = player.Character.HumanoidRootPart
			local dist = (targetRoot.Position - myPos).Magnitude
			
			if dist <= closestDist then
				closestDist = dist
				closestTarget = targetRoot
			end
		end
	end

	return closestTarget
end

-- Target casuale (Slap All)
local function getRandomTarget()
	local validTargets = {}
	for _, player in pairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			table.insert(validTargets, player.Character.HumanoidRootPart)
		end
	end
	
	if #validTargets > 0 then
		return validTargets[math.random(1, #validTargets)]
	end
	return nil
end

-- Controlla se il player ha un tool nello zaino o equipaggiato
local function hasTool()
	local character = LocalPlayer.Character
	local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
	
	if character and character:FindFirstChildOfClass("Tool") then
		return true
	end
	if backpack and backpack:FindFirstChildOfClass("Tool") then
		return true
	end
	
	return false
end

-- Anti Fling Logic
local lastSafeCFrame = nil
local timeInAir = 0

task.spawn(function()
	while true do
		task.wait(0.1)
		local myChar = LocalPlayer.Character
		if myChar and myChar:FindFirstChild("HumanoidRootPart") and myChar:FindFirstChildOfClass("Humanoid") then
			local myRoot = myChar.HumanoidRootPart
			local humanoid = myChar:FindFirstChildOfClass("Humanoid")
			
			if humanoid.FloorMaterial ~= Enum.Material.Air then
				if not slapAllActive and not autoPlayingActive then
					lastSafeCFrame = myRoot.CFrame
				end
				timeInAir = 0
			else
				if antiFlingActive and not slapAllActive and not autoPlayingActive then
					timeInAir = timeInAir + 0.1
					if timeInAir >= 1.0 and lastSafeCFrame then
						myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
						myRoot.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
						myRoot.CFrame = lastSafeCFrame
						timeInAir = 0
					end
				end
			end
		end
	end
end)

-- Gestione Respawn / Morte per Auto Playing
LocalPlayer.CharacterAdded:Connect(function(newChar)
	task.wait(1)
	if autoPlayingActive then
		local tpPart = getTeleport1Part()
		local root = newChar:WaitForChild("HumanoidRootPart", 5)
		if tpPart and root then
			root.CFrame = tpPart.CFrame + Vector3.new(0, 3, 0)
		end
	end
end)

-- Loop Auto Slap
task.spawn(function()
	while true do
		task.wait(0.1)
		if autoSlapActive and not slapAllActive and not autoPlayingActive and not isExecuting then
			local myChar = LocalPlayer.Character
			if myChar and myChar:FindFirstChild("HumanoidRootPart") then
				local targetRoot = getClosestTarget()
				if targetRoot then
					isExecuting = true
					local myRoot = myChar.HumanoidRootPart
					local originalCFrame = myRoot.CFrame
					local humanoid = myChar:FindFirstChildOfClass("Humanoid")
					
					if humanoid then humanoid.PlatformStand = true end
					myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, DISTANZA_DIETRO)
					RunService.RenderStepped:Wait()
					
					simulateClicks(1)
					task.wait(0.08)
					
					myRoot.CFrame = originalCFrame
					if humanoid then humanoid.PlatformStand = false end
					
					task.wait(0.4)
					isExecuting = false
				end
			end
		end
	end
end)

-- Loop Slap All Classico
task.spawn(function()
	while true do
		task.wait(1)
		if slapAllActive and not autoPlayingActive then
			local myChar = LocalPlayer.Character
			if myChar and myChar:FindFirstChild("HumanoidRootPart") then
				local targetRoot = getRandomTarget()
				if targetRoot then
					local myRoot = myChar.HumanoidRootPart
					local humanoid = myChar:FindFirstChildOfClass("Humanoid")
					
					if humanoid then humanoid.PlatformStand = true end
					myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, DISTANZA_DIETRO)
					RunService.RenderStepped:Wait()
					
					simulateClicks(5)
					
					myRoot.CFrame = CFrame.new(platform.Position + Vector3.new(0, 3, 0))
					if humanoid then humanoid.PlatformStand = false end
				end
			end
		end
	end
end)

-- Loop Auto Playing
task.spawn(function()
	while true do
		task.wait(0.5)
		if autoPlayingActive then
			local myChar = LocalPlayer.Character
			if myChar and myChar:FindFirstChild("HumanoidRootPart") then
				local myRoot = myChar.HumanoidRootPart
				local tpPart = getTeleport1Part()
				
				if not hasTool() then
					if tpPart then
						myRoot.CFrame = tpPart.CFrame + Vector3.new(0, 3, 0)
					end
				else
					local targetRoot = getRandomTarget()
					if targetRoot and tpPart then
						local humanoid = myChar:FindFirstChildOfClass("Humanoid")
						if humanoid then humanoid.PlatformStand = true end
						
						myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, DISTANZA_DIETRO)
						RunService.RenderStepped:Wait()
						
						simulateClicks(5)
						
						myRoot.CFrame = tpPart.CFrame + Vector3.new(0, 3, 0)
						if humanoid then humanoid.PlatformStand = false end
					end
				end
			end
		end
	end
end)
