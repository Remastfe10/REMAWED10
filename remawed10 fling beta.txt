-- Nuova funzione di Fling basata sulla fisica di Roblox (Client-side)
local FLING_DURATION = 4.5
local SPIN_SPEED = 5500

local function getCharacterFromPart(part: Instance?): Model?
	if not part then
		return nil
	end

	local current: Instance? = part
	while current and current ~= Workspace do
		if current:IsA("Model") and current:FindFirstChildOfClass("Humanoid") then
			return current
		end
		current = current.Parent
	end

	return nil
end

local function setCharacterCollide(character: Model, enabled: boolean)
	for _, descendant in ipairs(character:GetDescendants()) do
		if descendant:IsA("BasePart") then
			descendant.CanCollide = enabled
		end
	end
end

local function resetCharacterPhysics(character: Model, root: BasePart)
	for _, descendant in ipairs(character:GetDescendants()) do
		if descendant:IsA("BasePart") then
			descendant.CustomPhysicalProperties = nil
			descendant.AssemblyLinearVelocity = Vector3.zero
			descendant.AssemblyAngularVelocity = Vector3.zero
		end
	end

	local oldSpin = root:FindFirstChild("remawed10_Spin")
	if oldSpin then
		oldSpin:Destroy()
	end
end

local function flingTarget(targetCharacter: Model)
	if isExecuting then
		return
	end

	local myCharacter = player.Character
	if not myCharacter or targetCharacter == myCharacter then
		return
	end

	local myRoot = myCharacter:FindFirstChild("HumanoidRootPart") :: BasePart?
	local myHumanoid = myCharacter:FindFirstChildOfClass("Humanoid")
	local targetRoot = targetCharacter:FindFirstChild("HumanoidRootPart") :: BasePart?
	local targetHumanoid = targetCharacter:FindFirstChildOfClass("Humanoid")

	if not myRoot or not myHumanoid or not targetRoot or not targetHumanoid then
		return
	end

	isExecuting = true
	local startTime = os.clock()
	local originalCFrame = myRoot.CFrame

	-- Mette a fuoco la telecamera sul bersaglio durante l'azione
	camera.CameraSubject = targetHumanoid

	-- Riduce l'attrito e attiva le collisioni per trasferire impulso fisico
	for _, part in ipairs(myCharacter:GetDescendants()) do
		if part:IsA("BasePart") then
			part.CustomPhysicalProperties = PhysicalProperties.new(0.01, 0, 0, 0, 0)
			part.CanCollide = true
		end
	end

	-- Rotazione ad alta velocità tramite BodyAngularVelocity
	local spinForce = Instance.new("BodyAngularVelocity")
	spinForce.Name = "remawed10_Spin"
	spinForce.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
	spinForce.AngularVelocity = Vector3.new(0, SPIN_SPEED, 0)
	spinForce.Parent = myRoot

	local stepConnection: RBXScriptConnection
	stepConnection = RunService.Heartbeat:Connect(function(dt)
		local elapsed = os.clock() - startTime
		local targetDead = targetHumanoid.Health <= 0 or targetHumanoid:GetState() == Enum.HumanoidStateType.Dead

		if elapsed >= FLING_DURATION or targetDead or not isActive or not targetRoot.Parent then
			stepConnection:Disconnect()

			-- Ferma l'inerzia e ripristina il personaggio
			resetCharacterPhysics(myCharacter, myRoot)
			setCharacterCollide(myCharacter, false)
			myRoot.CFrame = originalCFrame

			task.wait(0.05)
			setCharacterCollide(myCharacter, true)
			myRoot.CFrame = originalCFrame
			myRoot.AssemblyLinearVelocity = Vector3.zero
			myRoot.AssemblyAngularVelocity = Vector3.zero
			camera.CameraSubject = myHumanoid

			isExecuting = false
			return
		end

		-- Segue la posizione prevista del bersaglio e applica una spinta verso l'alto/avanti
		local predictedPos = targetRoot.Position + (targetRoot.AssemblyLinearVelocity * math.min(dt * 2, 0.15))
		local upwardBoost = Vector3.new(0, 140, 0)
		local outOffset = targetRoot.CFrame.LookVector * 1.2

		myRoot.CFrame = CFrame.new(predictedPos + outOffset) * CFrame.Angles(0, elapsed * 35, 0)
		myRoot.AssemblyLinearVelocity = targetRoot.AssemblyLinearVelocity + upwardBoost
	end)
end

mouse.Button1Down:Connect(function()
	if not isActive or isExecuting then
		return
	end

	local targetCharacter = getCharacterFromPart(mouse.Target)
	if targetCharacter then
		flingTarget(targetCharacter)
	end
end)
