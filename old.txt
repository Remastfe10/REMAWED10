-- === FUNZIONE 1: OLD AVATAR (R15 RESPAWN + POSA STATICA SENZA ANIMAZIONI) ===

local function cacheAnimateScript(character: Model)
	if savedAnimateTemplate then
		return
	end
	local animate = character:FindFirstChild("Animate")
	if animate and (animate:IsA("LocalScript") or animate:IsA("Script")) then
		savedAnimateTemplate = animate:Clone()
	end
end

local function unbindAvatarLock()
	pcall(function()
		RunService:UnbindFromRenderStep(AVATAR_BIND_NAME)
	end)
end

local function restoreAnimations(character: Model)
	unbindAvatarLock()

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	for _, desc in ipairs(character:GetDescendants()) do
		if desc:IsA("Motor6D") then
			desc.Transform = CFrame.new()
		end
	end

	if humanoid then
		local animator = humanoid:FindFirstChildOfClass("Animator")
		if not animator then
			animator = Instance.new("Animator")
			animator.Parent = humanoid
		end
	end

	local currentAnimate = character:FindFirstChild("Animate")
	if currentAnimate then
		if currentAnimate:IsA("LocalScript") or currentAnimate:IsA("Script") then
			currentAnimate.Enabled = true
		end
	elseif savedAnimateTemplate then
		local restored = savedAnimateTemplate:Clone()
		restored.Parent = character
		if restored:IsA("LocalScript") or restored:IsA("Script") then
			restored.Enabled = true
		end
	end
end

local function applyStaticR6Pose(character: Model, myToken: number)
	clearConnections(avatarConnections)
	unbindAvatarLock()
	cacheAnimateScript(character)

	local humanoid = character:FindFirstChild("Humanoid") :: Humanoid?
	if not humanoid then
		return
	end

	local function disableAnimateNow()
		local animateScript = character:FindFirstChild("Animate")
		if animateScript and (animateScript:IsA("LocalScript") or animateScript:IsA("Script")) then
			animateScript.Enabled = false
		end
	end

	disableAnimateNow()

	table.insert(avatarConnections, character.ChildAdded:Connect(function(child)
		if not oldAvatarEnabled or myToken ~= avatarToken then
			return
		end
		if child.Name == "Animate" and (child:IsA("LocalScript") or child:IsA("Script")) then
			child.Enabled = false
		end
	end))

	local function stopAllTracks(animator: Animator?)
		if not animator then
			return
		end
		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			track:Stop(0)
		end
	end

	local animator = humanoid:FindFirstChildOfClass("Animator")
	if animator then
		stopAllTracks(animator)
		table.insert(avatarConnections, animator.AnimationPlayed:Connect(function(track: AnimationTrack)
			if not oldAvatarEnabled or myToken ~= avatarToken then
				return
			end
			track:Stop(0)
		end))
	end

	local motors: { Motor6D } = {}
	for _, desc in ipairs(character:GetDescendants()) do
		if desc:IsA("Motor6D") then
			table.insert(motors, desc)
		end
	end

	table.insert(avatarConnections, character.DescendantAdded:Connect(function(desc)
		if not oldAvatarEnabled or myToken ~= avatarToken then
			return
		end
		if desc:IsA("Motor6D") then
			table.insert(motors, desc)
		end
	end))

	RunService:BindToRenderStep(AVATAR_BIND_NAME, Enum.RenderPriority.Character.Value + 1, function()
		if not oldAvatarEnabled or myToken ~= avatarToken or not character.Parent then
			return
		end

		disableAnimateNow()

		local curAnimator = humanoid:FindFirstChildOfClass("Animator")
		stopAllTracks(curAnimator)

		local isR15 = humanoid.RigType == Enum.HumanoidRigType.R15
		for _, motor in ipairs(motors) do
			if motor.Parent then
				if isR15 then
					motor.Transform = R15_STATIC_R6_OFFSETS[motor.Name] or CFrame.new()
				else
					motor.Transform = CFrame.new()
				end
			end
		end
	end)
end

local function activateOldAvatarFlow(myToken: number)
	local character = player.Character
	if not character then
		return
	end
	cacheAnimateScript(character)

	local root = character:FindFirstChild("HumanoidRootPart") :: BasePart?
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid then
		return
	end

	showNotification("OLD AVATAR", "Respawning into static R6 pose...", 2.5)
	task.wait(1.2)

	if not oldAvatarEnabled or myToken ~= avatarToken then
		return
	end

	if character.Parent and root.Parent and humanoid.Health > 0 then
		savedCFrame = root.CFrame
		waitingToRestorePosition = true
		humanoid.Health = 0
	end
end

local function disableOldAvatar()
	oldAvatarEnabled = false
	avatarToken += 1
	waitingToRestorePosition = false
	savedCFrame = nil
	clearConnections(avatarConnections)
	unbindAvatarLock()
	if player.Character then
		restoreAnimations(player.Character)
	end
end

local ALL_FACES: { Enum.NormalId } = {
	Enum.NormalId.Top,
	Enum.NormalId.Bottom,
	Enum.NormalId.Left,
	Enum.NormalId.Right,
	Enum.NormalId.Front,
	Enum.NormalId.Back,

local function getProceduralStudTexture(): string?
	if cachedProceduralStudContent then
		return cachedProceduralStudContent
	end

	local ok, content = pcall(function()
		local AssetService = game:GetService("AssetService")
		if not AssetService or not AssetService.CreateEditableImage then
			return nil
		end

		local size = 128
		local half = size // 2
		local img = AssetService:CreateEditableImage({ Size = Vector2.new(size, size) })
		local bufferData = table.create(size * size * 4, 0)

		local function writePixel(px: number, py: number, r: number, g: number, b: number, a: number)
			local idx = ((py * size) + px) * 4 + 1
			bufferData[idx] = math.clamp(r, 0, 1)
			bufferData[idx + 1] = math.clamp(g, 0, 1)
			bufferData[idx + 2] = math.clamp(b, 0, 1)
			bufferData[idx + 3] = math.clamp(a, 0, 1)
		end

		for y = 0, size - 1 do
			for x = 0, size - 1 do
				writePixel(x, y, 0.56, 0.56, 0.56, 0.65)
			end
		end

		for cellY = 0, 1 do
			for cellX = 0, 1 do
				local ox = cellX * half
				local oy = cellY * half
				local margin = 10
				local studSize = half - margin * 2
				local bevel = 5

				for dy = 0, studSize - 1 do
					for dx = 0, studSize - 1 do
						local px = ox + margin + dx
						local py = oy + margin + dy
						if px >= 0 and px < size and py >= 0 and py < size then
							local edgeDist = math.min(dx, dy, studSize - 1 - dx, studSize - 1 - dy)
							local shade = 0.72
							if edgeDist < bevel then
								local t = edgeDist / bevel
								local isHighlight = (dx < dy)
								shade = isHighlight and (0.58 + 0.24 * t) or (0.34 + 0.30 * t)
							else
								shade = 0.76
							end
							writePixel(px, py, shade, shade, shade, 0.92)
						end
					end
				end
			end
		end

		img:WritePixels(Vector2.zero, Vector2.new(size, size), bufferData)
		return img.ImageContent
	end)

	if ok and type(content) == "string" and content ~= "" then
		cachedProceduralStudContent = content
		return content
	end

	return nil
end

local function applyStudsToPart(part: BasePart, myToken: number)
	if not oldGraphicsEnabled or myToken ~= graphicsToken or not part:IsA("BasePart") then
		return
	end
	if part:FindFirstChild("ClassicStudApplied") then
		return
	end

	local marker = Instance.new("BoolValue")
	marker.Name = "ClassicStudApplied"
	marker.Parent = part

	local textureId = getProceduralStudTexture() or PRIMARY_STUD_ASSET

	for _, face in ipairs(ALL_FACES) do
		local tex = Instance.new("Texture")
		tex.Name = "OldRobloxStud"
		tex.Face = face
		tex.Texture = textureId
		tex.StudsPerTileU = STUD_TILE_SIZE
		tex.StudsPerTileV = STUD_TILE_SIZE
		tex.Transparency = 0.12
		tex.ZIndex = 10
		tex.Parent = part
		table.insert(appliedStuds, tex)
	end
end

local function applyStudsToContainer(container: Instance, myToken: number)
	if not oldGraphicsEnabled or myToken ~= graphicsToken then
		return
	end
	if container:IsA("BasePart") then
		applyStudsToPart(container, myToken)
	end
	for _, desc in ipairs(container:GetDescendants()) do
		if not oldGraphicsEnabled or myToken ~= graphicsToken then
			return
		end
		if desc:IsA("BasePart") then
			applyStudsToPart(desc, myToken)
		end
	end
end

local function disableOldGraphics()
	oldGraphicsEnabled = false
	graphicsToken += 1
	clearConnections(graphicsConnections)

	for _, tex in ipairs(appliedStuds) do
		if tex and tex.Parent then
			tex:Destroy()
		end
	end
	table.clear(appliedStuds)

	for _, desc in ipairs(Workspace:GetDescendants()) do
		if desc.Name == "ClassicStudApplied" and desc:IsA("BoolValue") then
			desc:Destroy()
		elseif desc.Name == "OldRobloxStud" and desc:IsA("Texture") then
			desc:Destroy()
		end
	end
end

local function enableOldGraphics()
	disableOldGraphics()
	oldGraphicsEnabled = true
	local myToken = graphicsToken

	task.spawn(function()
		local count = 0
		for _, desc in ipairs(Workspace:GetDescendants()) do
			if not oldGraphicsEnabled or myToken ~= graphicsToken then
				return
			end
			if desc:IsA("BasePart") then
				applyStudsToPart(desc, myToken)
				count += 1
				if count % 250 == 0 then
					RunService.Heartbeat:Wait()
				end
			end
		end

		for _, plr in ipairs(Players:GetPlayers()) do
			if not oldGraphicsEnabled or myToken ~= graphicsToken then
				return
			end
			if plr.Character then
				applyStudsToContainer(plr.Character, myToken)
			end
		end
	end)

	table.insert(graphicsConnections, Workspace.DescendantAdded:Connect(function(desc)
		if not oldGraphicsEnabled or myToken ~= graphicsToken then
			return
		end
		if desc:IsA("BasePart") then
			applyStudsToPart(desc, myToken)
		end
	end))
end

local CORE_GUI_TYPES: { Enum.CoreGuiType } = {

local function sweepAndHideModernUis()
	local playerGui = player:FindFirstChild("PlayerGui")
	if not playerGui then
		return
	end

	for _, gui in ipairs(playerGui:GetChildren()) do
		if gui:IsA("ScreenGui") and gui.Name ~= "OldAvatarPanelGui" and gui.Name ~= "OldRobloxClassicGui" then
			local lowerName = string.lower(gui.Name)
			if string.find(lowerName, "chat", 1, true)
				or string.find(lowerName, "topbar", 1, true)
				or string.find(lowerName, "emote", 1, true)
				or string.find(lowerName, "playerlist", 1, true)
				or string.find(lowerName, "unibar", 1, true)
			then
				for _, child in ipairs(gui:GetChildren()) do
					if child:IsA("GuiObject") and child.Visible then
						if hiddenGuiObjects[child] == nil then
							hiddenGuiObjects[child] = true
						end
						child.Visible = false
					end
				end
			end
		end
	end
end

local function setModernGuiVisibility(hide: boolean)
	if hide then
		for _, coreType in ipairs(CORE_GUI_TYPES) do
			pcall(function()
				if savedCoreGuiStates[coreType] == nil then
					savedCoreGuiStates[coreType] = StarterGui:GetCoreGuiEnabled(coreType)
				end
				StarterGui:SetCoreGuiEnabled(coreType, false)
			end)
		end

		pcall(function()
			local chatWindow = TextChatService:FindFirstChildOfClass("ChatWindowConfiguration")
			local chatInput = TextChatService:FindFirstChildOfClass("ChatInputBarConfiguration")
			local bubbleChat = TextChatService:FindFirstChildOfClass("BubbleChatConfiguration")

			if chatWindow and savedChatWindowEnabled == nil then
				savedChatWindowEnabled = chatWindow.Enabled
			end
			if chatInput and savedChatInputEnabled == nil then
				savedChatInputEnabled = chatInput.Enabled
			end
			if bubbleChat and savedBubbleChatEnabled == nil then
				savedBubbleChatEnabled = bubbleChat.Enabled
			end
			if chatWindow then
				chatWindow.Enabled = false
			end
			if chatInput then
				chatInput.Enabled = false
			end
			if bubbleChat then
				bubbleChat.Enabled = false
			end
		end)

		sweepAndHideModernUis()
	else
		for coreType, prevState in pairs(savedCoreGuiStates) do
			pcall(function()
				StarterGui:SetCoreGuiEnabled(coreType, prevState)
			end)
		end
		table.clear(savedCoreGuiStates)

		pcall(function()
			local chatWindow = TextChatService:FindFirstChildOfClass("ChatWindowConfiguration")
			local chatInput = TextChatService:FindFirstChildOfClass("ChatInputBarConfiguration")
			local bubbleChat = TextChatService:FindFirstChildOfClass("BubbleChatConfiguration")

			if chatWindow and savedChatWindowEnabled ~= nil then
				chatWindow.Enabled = savedChatWindowEnabled
			end
			if chatInput and savedChatInputEnabled ~= nil then
				chatInput.Enabled = savedChatInputEnabled
			end
			if bubbleChat and savedBubbleChatEnabled ~= nil then
				bubbleChat.Enabled = savedBubbleChatEnabled
			end
			savedChatWindowEnabled = nil
			savedChatInputEnabled = nil
			savedBubbleChatEnabled = nil
		end)
	end
end

-- === FUNZIONE 4: OLD AVATAR ALL (Tutti noob, senza roba/vestiti addosso, faccia classica smile) ===

local function getNoobColor(partName: string): Color3?
	local lower = string.lower(partName)
	if lower == "head" then
		return NOOB_HEAD_COLOR
	elseif string.find(lower, "torso", 1, true) then
		return NOOB_TORSO_COLOR
	elseif string.find(lower, "arm", 1, true) or string.find(lower, "hand", 1, true) then
		return NOOB_ARM_COLOR
	elseif string.find(lower, "leg", 1, true) or string.find(lower, "foot", 1, true) then
		return NOOB_LEG_COLOR
	end
	return nil
end

local function isCharacterModel(model: Instance): boolean
	return model:IsA("Model") and model:FindFirstChildOfClass("Humanoid") ~= nil
end

local function applyNoobAllToCharacter(character: Model, myToken: number)
	if not oldAvatarAllEnabled or myToken ~= avatarAllToken or not character.Parent then
		return
	end

	local backup = noobAllBackups[character]
	if not backup then
		backup = {
			Parts = {},
			Decals = {},
			Clothing = {},
			AddedFace = nil,
		}
		noobAllBackups[character] = backup
	end

	for _, child in ipairs(character:GetChildren()) do
		if child:IsA("Shirt") or child:IsA("Pants") or child:IsA("ShirtGraphic") or child:IsA("Accessory") then
			table.insert(backup.Clothing, {
				Instance = child,
				Parent = child.Parent,
			})
			child.Parent = nil
		end
	end

	for _, desc in ipairs(character:GetDescendants()) do
		if desc:IsA("BasePart") and desc.Name ~= "HumanoidRootPart" then
			local noobCol = getNoobColor(desc.Name)
			if noobCol then
				if not backup.Parts[desc] then
					backup.Parts[desc] = {
						Color = desc.Color,
						Material = desc.Material,
						Transparency = desc.Transparency,
					}
				end
				desc.Color = noobCol
				desc.Material = Enum.Material.SmoothPlastic
				desc.Transparency = 0
			end
		elseif desc:IsA("Decal") and desc.Name ~= "ClassicNoobSmile" then
			table.insert(backup.Decals, {
				Decal = desc,
				Texture = desc.Texture,
				Transparency = desc.Transparency,
			})
			desc.Transparency = 1
		end
	end

	local head = character:FindFirstChild("Head") :: BasePart?
	if head then
		local existing = head:FindFirstChild("ClassicNoobSmile")
		if existing and existing:IsA("Decal") then
			existing.Texture = NOOB_SMILE_FACE
			existing.Transparency = 0
			backup.AddedFace = existing
		elseif not backup.AddedFace then
			local face = Instance.new("Decal")
			face.Name = "ClassicNoobSmile"
			face.Face = Enum.NormalId.Front
			face.Texture = NOOB_SMILE_FACE
			face.Transparency = 0
			face.Parent = head
			backup.AddedFace = face
		end
	end
end

local function restoreNoobAllCharacters()
	for character, backup in pairs(noobAllBackups) do
		if backup.AddedFace and backup.AddedFace.Parent then
			backup.AddedFace:Destroy()
		end
		for _, cState in ipairs(backup.Clothing) do
			if cState.Instance and cState.Parent and character.Parent then
				cState.Instance.Parent = cState.Parent
			end
		end
		for part, state in pairs(backup.Parts) do
			if part and part.Parent then
				part.Color = state.Color
				part.Material = state.Material
				part.Transparency = state.Transparency
			end
		end
		for _, dState in ipairs(backup.Decals) do
			if dState.Decal and dState.Decal.Parent then
				dState.Decal.Texture = dState.Texture
				dState.Decal.Transparency = dState.Transparency
			end
		end
	end
	table.clear(noobAllBackups)
end

local function disableOldAvatarAll()
	oldAvatarAllEnabled = false
	avatarAllToken += 1
	clearConnections(avatarAllConnections)
	restoreNoobAllCharacters()
end

local function enableOldAvatarAll()
	disableOldAvatarAll()
	oldAvatarAllEnabled = true
	local myToken = avatarAllToken

	if player.Character then
		applyNoobAllToCharacter(player.Character, myToken)
	end

	for _, child in ipairs(Workspace:GetChildren()) do
		if isCharacterModel(child) then
			applyNoobAllToCharacter(child :: Model, myToken)
		end
	end

	table.insert(avatarAllConnections, Workspace.ChildAdded:Connect(function(child)
		if not oldAvatarAllEnabled or myToken ~= avatarAllToken then
			return
		end
		if child:IsA("Model") then
			task.delay(0.15, function()
				if oldAvatarAllEnabled and myToken == avatarAllToken and isCharacterModel(child) then
					applyNoobAllToCharacter(child, myToken)
				end
			end)
		end
	end))
end

-- === FUNZIONE 5: NO ANIM ALL ===

local function applyNoAnimToOtherCharacter(character: Model, myToken: number)
	if not noAnimAllEnabled or myToken ~= noAnimAllToken or character == player.Character or not character.Parent then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	local backup = noAnimAllBackups[character]
	if not backup then
		backup = {
			DisabledScripts = {},
			PoseConn = nil,
		}
		noAnimAllBackups[character] = backup
	end

	for _, desc in ipairs(character:GetDescendants()) do
		if (desc:IsA("LocalScript") or desc:IsA("Script")) and desc.Name == "Animate" and desc.Enabled then
			desc.Enabled = false
			table.insert(backup.DisabledScripts, desc)
		end
	end

	local animator = humanoid:FindFirstChildOfClass("Animator")
	if animator then
		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			track:Stop(0)
		end
	end

	if not backup.PoseConn then
		local motors: { Motor6D } = {}
		for _, desc in ipairs(character:GetDescendants()) do
			if desc:IsA("Motor6D") then
				table.insert(motors, desc)
			end
		end

		backup.PoseConn = RunService.Stepped:Connect(function()
			if not noAnimAllEnabled or myToken ~= noAnimAllToken or not character.Parent then
				return
			end
			local curAnimator = humanoid:FindFirstChildOfClass("Animator")
			if curAnimator then
				for _, track in ipairs(curAnimator:GetPlayingAnimationTracks()) do
					track:Stop(0)
				end
			end
			local isR15 = humanoid.RigType == Enum.HumanoidRigType.R15
			for _, m in ipairs(motors) do
				if m.Parent then
					if isR15 then
						m.Transform = R15_STATIC_R6_OFFSETS[m.Name] or CFrame.new()
					else
						m.Transform = CFrame.new()
					end
				end
			end
		end)
	end
end

local function restoreNoAnimAllCharacters()
	for character, backup in pairs(noAnimAllBackups) do
		if backup.PoseConn then
			backup.PoseConn:Disconnect()
			backup.PoseConn = nil
		end
		for _, scr in ipairs(backup.DisabledScripts) do
			if scr and scr.Parent then
				scr.Enabled = true
			end
		end
		if character and character.Parent then
			for _, desc in ipairs(character:GetDescendants()) do
				if desc:IsA("Motor6D") then
					desc.Transform = CFrame.new()
				end
			end
		end
	end
	table.clear(noAnimAllBackups)
end

local function disableNoAnimAll()
	noAnimAllEnabled = false
	noAnimAllToken += 1
	clearConnections(noAnimAllConnections)
	restoreNoAnimAllCharacters()
end

local function enableNoAnimAll()
	disableNoAnimAll()
	noAnimAllEnabled = true
	local myToken = noAnimAllToken

	for _, child in ipairs(Workspace:GetChildren()) do
		if isCharacterModel(child) and child ~= player.Character then
			applyNoAnimToOtherCharacter(child :: Model, myToken)
		end
	end

	table.insert(noAnimAllConnections, Workspace.ChildAdded:Connect(function(child)
		if not noAnimAllEnabled or myToken ~= noAnimAllToken then
			return
		end
		if child:IsA("Model") and child ~= player.Character then
			task.delay(0.15, function()
				if noAnimAllEnabled and myToken == noAnimAllToken and isCharacterModel(child) then
					applyNoAnimToOtherCharacter(child, myToken)
				end
			end)
		end
	end))
end

-- === FUNZIONE 6: OLD JUMP SOUND (129512638771628) ===

local function playOldJumpSound(parentTarget: Instance?)
	local sound = Instance.new("Sound")
	sound.Name = "OldJumpSoundEffect"
	sound.SoundId = OLD_JUMP_SOUND_ID
	sound.Volume = 0.85
	sound.Parent = parentTarget or SoundService
	sound:Play()
	sound.Ended:Connect(function()
		if sound and sound.Parent then
			sound:Destroy()
		end
	end)
	task.delay(5, function()
		if sound and sound.Parent then
			sound:Destroy()
		end
	end)
end

local function hookCharacterJumpSound(character: Model, myToken: number)
	if not oldJumpSoundEnabled or myToken ~= jumpSoundToken or not character.Parent then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local root = character:FindFirstChild("HumanoidRootPart")

	local function patchJumpingSounds(container: Instance)
		for _, desc in ipairs(container:GetDescendants()) do
			if desc:IsA("Sound") and string.lower(desc.Name) == "jumping" then
				local alreadySaved = false
				for _, item in ipairs(jumpSoundBackups) do
					if item.Sound == desc then
						alreadySaved = true
						break
					end
				end
				if not alreadySaved then
					table.insert(jumpSoundBackups, {
						Sound = desc,
						SoundId = desc.SoundId,
					})
				end
				desc.SoundId = OLD_JUMP_SOUND_ID
			end
		end
	end

	patchJumpingSounds(character)

	table.insert(jumpSoundConnections, character.DescendantAdded:Connect(function(desc)
		if not oldJumpSoundEnabled or myToken ~= jumpSoundToken then
			return
		end
		if desc:IsA("Sound") and string.lower(desc.Name) == "jumping" then
			table.insert(jumpSoundBackups, {
				Sound = desc,
				SoundId = desc.SoundId,
			})
			desc.SoundId = OLD_JUMP_SOUND_ID
		end
	end))

	local function triggerJumpAudio()
		if not oldJumpSoundEnabled or myToken ~= jumpSoundToken then
			return
		end
		local now = os.clock()
		if now - lastJumpTriggerTime < 0.12 then
			return
		end
		lastJumpTriggerTime = now
		playOldJumpSound(root or character)
	end

	if humanoid then
		table.insert(jumpSoundConnections, humanoid.Jumping:Connect(function(isJumping: boolean)
			if isJumping then
				triggerJumpAudio()
			end
		end))

		table.insert(jumpSoundConnections, humanoid.StateChanged:Connect(function(_, newState: Enum.HumanoidStateType)
			if newState == Enum.HumanoidStateType.Jumping then
				triggerJumpAudio()
			end
		end))
	end
end

local function disableOldJumpSound()
	oldJumpSoundEnabled = false
	jumpSoundToken += 1
	clearConnections(jumpSoundConnections)

	for _, item in ipairs(jumpSoundBackups) do
		if item.Sound and item.Sound.Parent then
			item.Sound.SoundId = item.SoundId
		end
	end
	table.clear(jumpSoundBackups)
end

local function enableOldJumpSound()
	disableOldJumpSound()
	oldJumpSoundEnabled = true
	local myToken = jumpSoundToken

	if player.Character then
		hookCharacterJumpSound(player.Character, myToken)
	end
end

-- === DRAG UI ===

local function makeDraggable(guiObject: GuiObject)
	local dragging = false
	local dragInput: InputObject? = nil
	local dragStart = Vector3.zero
	local startPos = UDim2.new()
	local moved = false

	guiObject.InputBegan:Connect(function(input: InputObject)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			moved = false
			dragStart = input.Position
			startPos = guiObject.Position
			dragInput = input

			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
					dragInput = nil
				end
			end)
		end
	end)

	guiObject.InputChanged:Connect(function(input: InputObject)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)

	UserInputService.InputChanged:Connect(function(input: InputObject)
		if dragging and input == dragInput then
			local delta = input.Position - dragStart
			if delta.Magnitude > 6 then
				moved = true
			end
			guiObject.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end)

	return function(): boolean
		local wasMoved = moved
		moved = false
		return wasMoved
	end
end

local function onCharacterAdded(character: Model)
	cacheAnimateScript(character)
	local root = character:WaitForChild("HumanoidRootPart") :: BasePart
	character:WaitForChild("Humanoid")

	task.wait(0.12)

	if waitingToRestorePosition and savedCFrame and oldAvatarEnabled then
		waitingToRestorePosition = false
		if character.Parent and root.Parent then
			character:PivotTo(savedCFrame)
		end
	else
		waitingToRestorePosition = false
		savedCFrame = nil
	end

	if oldAvatarEnabled then
		applyStaticR6Pose(character, avatarToken)
	else
		restoreAnimations(character)
	end

	if oldGraphicsEnabled then
		applyStudsToContainer(character, graphicsToken)
	end

	if oldAvatarAllEnabled then
		applyNoobAllToCharacter(character, avatarAllToken)
	end

	if oldJumpSoundEnabled then
		hookCharacterJumpSound(character, jumpSoundToken)
	end

	if updateButtonsUI then
		updateButtonsUI()
	end
end
