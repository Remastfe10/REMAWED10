-- FUNZIONE ESECUZIONE SCRIPT

local function executeSelected()
    if not Main.Visible then return end
    if not entries[selected] then return end
    local data = entries[selected]
    Status.Text = "ESEGUENDO: "..data.title
    
    local success, err = pcall(function()
        loadstring(data.code)()
    end)
    
    if not success then
        Status.Text = "ERRORE NELL'ESECUZIONE"
        warn("Remawed10 Error: "..tostring(err))
    end
end

-- SCROLL AUTOMATICO

local function scrollToSelected()
    if not Main.Visible then return end
    local pos = table.find(visibleEntries, selected)
    if not pos then return end
    local target = (pos-1)*68 -- Altezza ridotta della riga (senza descrizione)
    local maxScroll = math.max(0, Scroll.AbsoluteCanvasSize.Y - Scroll.AbsoluteSize.Y)
    target = math.clamp(target, 0, maxScroll)

    TweenService:Create(
        Scroll,
        TweenInfo.new(0.18, Enum.EasingStyle.Quad),
        {CanvasPosition = Vector2.new(0, target)}
    ):Play()
end

-- AGGIORNAMENTO LISTA

local function refresh()
    if not Main.Visible then return end
    for _,v in ipairs(Scroll:GetChildren()) do
        if v:IsA("Frame") then
            v:Destroy()
        end
    end

    visibleEntries = {}
    local query = Search.Text:lower()

    for i,data in ipairs(entries) do
        local matches = data.title:lower():find(query,1,true)

        if matches then
            table.insert(visibleEntries, i)

            local Row = Instance.new("Frame")
            Row.Size = UDim2.new(1,-8,0,58) -- Riga più compatta
            Row.BackgroundColor3 = (selected == i) and Color3.fromRGB(25,48,43) or PANEL
            Row.LayoutOrder = #visibleEntries
            Row.Parent = Scroll

            Instance.new("UICorner",Row).CornerRadius = UDim.new(0,10)

            local Border = Instance.new("UIStroke",Row)
            Border.Color = (selected == i) and GREEN or Color3.fromRGB(40,45,55)
            Border.Thickness = (selected == i) and 1.5 or 1

            local Num = Instance.new("TextLabel")
            Num.Size = UDim2.fromOffset(45,48)
            Num.Position = UDim2.fromOffset(10,5)
            Num.BackgroundTransparency = 1
            Num.Text = tostring(i)
            Num.TextColor3 = GREEN
            Num.Font = Enum.Font.GothamBlack
            Num.TextSize = 22
            Num.Parent = Row

            local Name = Instance.new("TextLabel")
            Name.Size = UDim2.new(1,-70,1,0)
            Name.Position = UDim2.fromOffset(60,0)
            Name.BackgroundTransparency = 1
            Name.Text = data.title
            Name.TextColor3 = WHITE
            Name.Font = Enum.Font.GothamBold
            Name.TextSize = 16
            Name.TextXAlignment = Enum.TextXAlignment.Left
            Name.Parent = Row

            local Select = Instance.new("TextButton")
            Select.Size = UDim2.new(1,0,1,0)
            Select.BackgroundTransparency = 1
            Select.Text = ""
            Select.Parent = Row

            Select.MouseButton1Click:Connect(function()
                if not Main.Visible then return end
                selected = i
                Status.Text = "SELEZIONATO: "..data.title
                refresh()
                scrollToSelected()
                executeSelected()
            end)
        end
    end

    task.defer(scrollToSelected)
end

-- MAPPATURA NUMERICA TASTIERA
local keyToNum = {
    [Enum.KeyCode.Zero] = "0", [Enum.KeyCode.One] = "1", [Enum.KeyCode.Two] = "2",
    [Enum.KeyCode.Three] = "3", [Enum.KeyCode.Four] = "4", [Enum.KeyCode.Five] = "5",
    [Enum.KeyCode.Six] = "6", [Enum.KeyCode.Seven] = "7", [Enum.KeyCode.Eight] = "8",
    [Enum.KeyCode.Nine] = "9",
    [Enum.KeyCode.KeypadZero] = "0", [Enum.KeyCode.KeypadOne] = "1", [Enum.KeyCode.KeypadTwo] = "2",
    [Enum.KeyCode.KeypadThree] = "3", [Enum.KeyCode.KeypadFour] = "4", [Enum.KeyCode.KeypadFive] = "5",
    [Enum.KeyCode.KeypadSix] = "6", [Enum.KeyCode.KeypadSeven] = "7", [Enum.KeyCode.KeypadEight] = "8",
    [Enum.KeyCode.KeypadNine] = "9"
}

-- CONTROLLI DA TASTIERA GLOBALI

UIS.InputBegan:Connect(function(input, processed)
    if input.KeyCode == Enum.KeyCode.G and not Search:IsFocused() then
        if Main.Visible then
            closePanel()
        else
            openPanel()
        end
        return
    end

    if not Main.Visible then return end
    if Search:IsFocused() then return end

    -- GESTIONE NUMERI DIRETTI
    local digit = keyToNum[input.KeyCode]
    if digit then
        local now = tick()
        if now - lastInputTime > 1.2 then
            inputBuffer = digit
        else
            inputBuffer = inputBuffer .. digit
        end
        lastInputTime = now

        local n = tonumber(inputBuffer)
        NumberBox.Text = "NUMERO: " .. inputBuffer

        if n and entries[n] then
            local found = table.find(visibleEntries, n)
            if found then
                selected = n
                Status.Text = "SELEZIONATO: " .. entries[n].title
                refresh()
                scrollToSelected()
            else
                Status.Text = "SLOT NON VISIBILE"
            end
        else
            Status.Text = "SLOT NON ESISTENTE"
        end
        return
    end

    -- ALTRI TASTI DI CONTROLLO
    if input.KeyCode == Enum.KeyCode.Down then
        local pos = table.find(visibleEntries, selected)
        if pos and pos < #visibleEntries then
            selected = visibleEntries[pos+1]
        elseif #visibleEntries > 0 then
            selected = visibleEntries[1]
        end
        refresh()
        scrollToSelected()

    elseif input.KeyCode == Enum.KeyCode.Up then
        local pos = table.find(visibleEntries, selected)
        if pos and pos > 1 then
            selected = visibleEntries[pos-1]
        elseif #visibleEntries > 0 then
            selected = visibleEntries[#visibleEntries]
        end
        refresh()
        scrollToSelected()

    elseif input.KeyCode == Enum.KeyCode.Return or input.KeyCode == Enum.KeyCode.KeypadEnter then
        executeSelected()

    elseif input.KeyCode == Enum.KeyCode.Escape then
        closePanel()
    end
end)
