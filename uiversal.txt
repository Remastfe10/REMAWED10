local function getCharacterParts()
	local char = localPlayer.Character

local function getTrueGameSpawnCFrame()
	local spawnLocations = {}
	for _, obj in ipairs(Workspace:GetDescendants()) do
		if obj:IsA("SpawnLocation") then
			table.insert(spawnLocations, obj)
		end
	end

	local chosenSpawn = nil
	if #spawnLocations > 0 then
		if localPlayer.Team then
			for _, sp in ipairs(spawnLocations) do
				if not sp.Neutral and sp.TeamColor == localPlayer.TeamColor then
					chosenSpawn = sp
					break
				end
			end
		end
		if not chosenSpawn then
			for _, sp in ipairs(spawnLocations) do
				if sp.Neutral then
					chosenSpawn = sp
					break
				end
			end
		end
		if not chosenSpawn then
			chosenSpawn = spawnLocations[1]
		end
	end

	if chosenSpawn then
		local yOffset = (chosenSpawn.Size.Y * 0.5) + 3.2
		return chosenSpawn.CFrame * CFrame.new(0, yOffset, 0)
	end

	return CFrame.new(0, 5, 0)
end

-- ==================== CLICK TELEPORT TOOL ====================
local hasClickTeleportTool = false

local function getClickWorldPosition()
	local camera = Workspace.CurrentCamera
	if not camera then return nil end

	local pointerPos = UserInputService:GetMouseLocation()
	local ray = camera:ViewportPointToRay(pointerPos.X, pointerPos.Y)

	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	local excludeList = {}
	if localPlayer.Character then
		table.insert(excludeList, localPlayer.Character)
	end
	rayParams.FilterDescendantsInstances = excludeList

	local result = Workspace:Raycast(ray.Origin, ray.Direction * 2500, rayParams)
	if result then
		return result.Position
	end

	return nil
end

local function createClickTeleportTool()
	local tool = Instance.new("Tool")
	tool.Name = "Click Teleport"
	tool.RequiresHandle = false
	tool.CanBeDropped = true
	tool.ToolTip = "Equip and click on a part to teleport"

	local isEquipped = false

	tool.Equipped:Connect(function()
		isEquipped = true
	end)

	tool.Unequipped:Connect(function()
		isEquipped = false
	end)

	tool.Activated:Connect(function()
		if not isEquipped or tool.Parent ~= localPlayer.Character then
			return
		end

		local char, hum, root = getCharacterParts()
		if not char or not root then return end

		local hitPos = getClickWorldPosition()
		if not hitPos then return end

		if hum and hum.Sit then
			hum.Sit = false
		end

		local targetPos = hitPos + Vector3.new(0, 3.2, 0)
		local lookVector = root.CFrame.LookVector
		local flatLook = Vector3.new(lookVector.X, 0, lookVector.Z)
		if flatLook.Magnitude < 0.01 then
			flatLook = Vector3.new(0, 0, -1)
		else
			flatLook = flatLook.Unit
		end

		root.AssemblyLinearVelocity = Vector3.zero
		root.AssemblyAngularVelocity = Vector3.zero
		char:PivotTo(CFrame.lookAt(targetPos, targetPos + flatLook))
	end)

	return tool
end

local function findLadderTopPosition(char, root)
	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = { char }

	local origin = root.Position
	local lookDir = root.CFrame.LookVector

	local overlapParams = OverlapParams.new()
	overlapParams.FilterType = Enum.RaycastFilterType.Exclude
	overlapParams.FilterDescendantsInstances = { char }

	local nearbyParts = Workspace:GetPartBoundsInRadius(origin, 6, overlapParams)
	for _, part in ipairs(nearbyParts) do
		if part:IsA("BasePart") then
			local lowerName = string.lower(part.Name)
			local isLadderLike = part:IsA("TrussPart")
				or string.find(lowerName, "ladder", 1, true) ~= nil
				or string.find(lowerName, "scale", 1, true) ~= nil

			if isLadderLike then
				local topY = part.Position.Y + (part.Size.Y * 0.5) + 3.5
				return Vector3.new(part.Position.X, topY, part.Position.Z) + (lookDir * 2.5)
			end
		end
	end

	local upRay = Workspace:Raycast(origin + lookDir * 1.5, Vector3.new(0, 25, 0), params)
	if upRay then
		return upRay.Position + Vector3.new(0, 3, 0) + (lookDir * 2)
	end

	return origin + Vector3.new(0, 12, 0) + (lookDir * 4)
end

local function trySkipLadder()
	local now = os.clock()
	if now - lastSkipTime < 0.35 then return end

	local char, hum, root = getCharacterParts()
	if not char or not hum or not root then return end

	local isClimbing = (hum:GetState() == Enum.HumanoidStateType.Climbing)
	local isMovingForward = hum.MoveDirection.Magnitude > 0.1

	if isClimbing and isMovingForward then
		lastSkipTime = now
		local targetPos = findLadderTopPosition(char, root)
		root.AssemblyLinearVelocity = Vector3.zero
		root.CFrame = CFrame.new(targetPos, targetPos + root.CFrame.LookVector)
		hum:ChangeState(Enum.HumanoidStateType.Running)
	end
end

local XRAY_TRANSPARENCY = 0.5

local function applyXrayToPart(part)
	if not part:IsA("BasePart") then return end
	if isCharacterPart(part) then return end

	if originalPartsTransparency[part] == nil then
		originalPartsTransparency[part] = part.Transparency
	end

	part.Transparency = math.max(originalPartsTransparency[part], XRAY_TRANSPARENCY)
end

local function getHumanoidAnchorPart(character)
	if not character then return nil end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if humanoid and humanoid.RootPart then
		return humanoid.RootPart
	end
	return character:FindFirstChild("HumanoidRootPart")
		or character:FindFirstChild("UpperTorso")
		or character:FindFirstChild("Torso")
		or character:FindFirstChildWhichIsA("BasePart")
end

local function ensureLocalAttachment()
	if localAttachment then
		localAttachment:Destroy()
		localAttachment = nil
	end

	local char = localPlayer.Character
	local anchor = getHumanoidAnchorPart(char)
	if anchor then
		local att = Instance.new("Attachment")
		att.Name = "Remawed10_LocalLineAtt"
		att.Parent = anchor
		localAttachment = att
	end

	for _, entry in pairs(ultraEspEntries) do
		if entry.beam then
			entry.beam.Attachment0 = localAttachment
			entry.beam.Enabled = (localAttachment ~= nil and entry.targetAttachment ~= nil)
		end
	end
end

local function clearUltraEspForPlayer(targetPlayer)
	local entry = ultraEspEntries[targetPlayer]
	if not entry then return end

	if entry.charConn then
		entry.charConn:Disconnect()
	end
	if entry.beam then
		entry.beam:Destroy()
	end
	if entry.targetAttachment then
		entry.targetAttachment:Destroy()
	end
	if entry.highlight then
		entry.highlight:Destroy()
	end

	ultraEspEntries[targetPlayer] = nil
end

local function attachUltraEspToTarget(targetPlayer, character)
	if not toggleStates["10. Ultra ESP Players"] or targetPlayer == localPlayer then return end

	local entry = ultraEspEntries[targetPlayer]
	if entry then
		if entry.beam then entry.beam:Destroy() end
		if entry.targetAttachment then entry.targetAttachment:Destroy() end
		if entry.highlight then entry.highlight:Destroy() end
	else
		entry = {}
		ultraEspEntries[targetPlayer] = entry
	end

	local humanoid = character and character:WaitForChild("Humanoid", 10)
	local anchor = humanoid and getHumanoidAnchorPart(character)
	if not anchor or not toggleStates["10. Ultra ESP Players"] then return end

	entry.highlight = createGreenBodyHighlight(character, "Remawed10_UltraESPBody")

	local targetAtt = Instance.new("Attachment")
	targetAtt.Name = "Remawed10_TargetLineAtt"
	targetAtt.Parent = anchor

	local beam = Instance.new("Beam")
	beam.Name = "Remawed10_UltraESPLine"
	beam.Attachment0 = localAttachment
	beam.Attachment1 = targetAtt
	beam.Color = ColorSequence.new(GREEN_COLOR)
	beam.LightEmission = 1
	beam.LightInfluence = 0
	beam.Transparency = NumberSequence.new(0)
	beam.Width0 = 0.14
	beam.Width1 = 0.14
	beam.FaceCamera = true
	beam.Enabled = (localAttachment ~= nil)
	beam.Parent = anchor

	entry.targetAttachment = targetAtt
	entry.beam = beam
end

local function createLiveStatRow(labelText, placeholderText, orderIndex, onValidLiveInput)
	local row = Instance.new("Frame")
	row.Name = labelText:gsub("%s+", "_") .. "_Row"
	row.Size = UDim2.new(1, 0, 0, 42)
	row.BackgroundColor3 = Color3.fromRGB(34, 38, 48)
	row.BorderSizePixel = 0
	row.LayoutOrder = orderIndex
	row.Parent = contentFrame

	local rowCorner = Instance.new("UICorner")
	rowCorner.CornerRadius = UDim.new(0, 8)
	rowCorner.Parent = row

	local rowStroke = Instance.new("UIStroke")
	rowStroke.Color = Color3.fromRGB(55, 60, 76)
	rowStroke.Thickness = 1
	rowStroke.Parent = row

	local textLabel = Instance.new("TextLabel")
	textLabel.Size = UDim2.new(1, -130, 1, 0)
	textLabel.Position = UDim2.new(0, 14, 0, 0)
	textLabel.BackgroundTransparency = 1
	textLabel.Text = labelText
	textLabel.TextColor3 = Color3.fromRGB(235, 238, 245)
	textLabel.TextSize = 13
	textLabel.Font = Enum.Font.GothamSemibold
	textLabel.TextXAlignment = Enum.TextXAlignment.Left
	textLabel.Parent = row

	local box = Instance.new("TextBox")
	box.Name = "ValueBox"
	box.Size = UDim2.new(0, 100, 0, 28)
	box.Position = UDim2.new(1, -110, 0.5, -14)
	box.BackgroundColor3 = Color3.fromRGB(22, 25, 33)
	box.Text = placeholderText
	box.PlaceholderText = placeholderText
	box.ClearTextOnFocus = false
	box.TextColor3 = Color3.fromRGB(255, 255, 255)
	box.PlaceholderColor3 = Color3.fromRGB(130, 138, 156)
	box.TextSize = 13
	box.Font = Enum.Font.GothamBold
	box.BorderSizePixel = 0
	box.Parent = row

	local boxCorner = Instance.new("UICorner")
	boxCorner.CornerRadius = UDim.new(0, 6)
	boxCorner.Parent = box

	local boxStroke = Instance.new("UIStroke")
	boxStroke.Color = Color3.fromRGB(85, 115, 215)
	boxStroke.Thickness = 1
	boxStroke.Parent = box

	box:GetPropertyChangedSignal("Text"):Connect(function()
		if syncingInputs then return end
		local cleaned = box.Text:gsub(",", ".")
		local num = tonumber(cleaned)
		if num and num >= 0 then
			onValidLiveInput(num)
		end
	end)

	box.FocusLost:Connect(function()
		local cleaned = box.Text:gsub(",", ".")
		local num = tonumber(cleaned)
		if not num or num < 0 then
			local _, hum = getCharacterParts()
			if hum then
				syncCharacterInputs(hum)
			end
		end
	end)

	return box
end

-- ==================== RESPAWN & MINIMIZE ====================
local function captureDefaultsFromHumanoid(hum)
	if not hum then return end
	defaultWalkSpeed = hum.WalkSpeed
	defaultJumpPower = hum.UseJumpPower and hum.JumpPower or 50
	if not hasCustomWalkSpeed then
		customWalkSpeed = defaultWalkSpeed
	end
	if not hasCustomJumpPower then
		customJumpPower = defaultJumpPower
	end
	applyCustomStatsToHumanoid(hum)
	syncCharacterInputs(hum)
end

if localPlayer.Character then
	local hum = localPlayer.Character:FindFirstChildOfClass("Humanoid")
	if hum then
		captureDefaultsFromHumanoid(hum)
	end
end

localPlayer.CharacterAdded:Connect(function(newChar)
	local hum = newChar:WaitForChild("Humanoid")
	task.wait(0.1)
	captureDefaultsFromHumanoid(hum)

	if hasClickTeleportTool then
		giveClickTeleportToolToBackpack()
	end

	for name, isOn in pairs(toggleStates) do
		if isOn and toggleSetters[name] then
			local setter = toggleSetters[name]
			setter(false, true)
			setter(true, true)
		end
	end
end)
