-- ==========================================
-- RILEVAMENTO SCALE E GRADINI
-- ==========================================
local function isStairInstance(inst)
	if not inst then
		return false
	end
	if inst:IsA("Seat") or inst:IsA("VehicleSeat") or inst:IsA("TrussPart") or inst:IsA("WedgePart") or inst:IsA("CornerWedgePart") then
		return true
	end
	local lowerName = string.lower(inst.Name)
	if string.find(lowerName, "stair")
		or string.find(lowerName, "step")
		or string.find(lowerName, "ramp")
		or string.find(lowerName, "ladder")
		or string.find(lowerName, "scala")
		or string.find(lowerName, "gradin") then
		return true
	end
	return false
end

local function isUsingStairs(character, humanoid, rootPart)
	if humanoid:GetState() == Enum.HumanoidStateType.Climbing then
		return true
	end

	local rayParams = RaycastParams.new()
	rayParams.FilterType = Enum.RaycastFilterType.Exclude
	rayParams.FilterDescendantsInstances = { character }

	local origin = rootPart.Position
	local look = rootPart.CFrame.LookVector

	local directDown = workspace:Raycast(origin, Vector3.new(0, -4.2, 0), rayParams)
	if directDown and directDown.Instance then
		if isStairInstance(directDown.Instance) or directDown.Distance <= 3.6 then
			return true
		end
	end

	local hit1 = workspace:Raycast(origin + look * 1.0, Vector3.new(0, -5.0, 0), rayParams)
	local hit2 = workspace:Raycast(origin + look * 2.2, Vector3.new(0, -5.5, 0), rayParams)
	if hit1 and hit2 then
		local heightDiff = math.abs(hit1.Position.Y - hit2.Position.Y)
		if heightDiff >= 0.15 and heightDiff <= 2.2 then
			return true
		end
	end

	local backHit = workspace:Raycast(origin - look * 1.2, Vector3.new(0, -4.5, 0), rayParams)
	if backHit and backHit.Distance <= 3.8 then
		return true
	end

	return false
end

-- ==========================================
-- LOGICA ANTI-FLING (TELEPORT PER ULTIMO)
-- ==========================================
local function teleportToSafeGround(rootPart, humanoid)
	rootPart.AssemblyLinearVelocity = Vector3.zero
	rootPart.AssemblyAngularVelocity = Vector3.zero
	humanoid:ChangeState(Enum.HumanoidStateType.Running)

	fallTime = 0
	inSuspiciousAir = false
	isJumping = false
	jumpLockTimer = 0
	teleportCooldown = 0.1

	local targetCFrame = previousSafeGroundCFrame or safeGroundCFrame
	if targetCFrame then
		rootPart.CFrame = targetCFrame + Vector3.new(0, 0.3, 0)
	end

	showXenoNotification(
		"Remawed10 ANTI-FLING",
		"Anti-Fling triggered! Returned to last safe position."
	)
end

local currentConn = nil

local function setupCharacter(character)
	if currentConn then
		currentConn:Disconnect()
		currentConn = nil
	end

	local humanoid = character:WaitForChild("Humanoid")
	local rootPart = character:WaitForChild("HumanoidRootPart")

	safeGroundCFrame = rootPart.CFrame
	previousSafeGroundCFrame = rootPart.CFrame
	lastSaveTime = os.clock()
	fallTime = 0
	inSuspiciousAir = false
	isJumping = false
	jumpLockTimer = 0
	teleportCooldown = 0

	humanoid.Jumping:Connect(function(active)
		if active then
			registerJump()
		end
	end)

	currentConn = RunService.Heartbeat:Connect(function(dt)
		if not character or not character.Parent or humanoid.Health <= 0 then
			if currentConn then
				currentConn:Disconnect()
				currentConn = nil
			end
			return
		end

		if teleportCooldown > 0 then
			teleportCooldown = math.max(0, teleportCooldown - dt)
			return
		end

		if jumpLockTimer > 0 then
			jumpLockTimer = math.max(0, jumpLockTimer - dt)
			if jumpLockTimer <= 0 then
				isJumping = false
			end
		end

		local vel = rootPart.AssemblyLinearVelocity
		local speed = vel.Magnitude
		local horizontalSpeed = Vector3.new(vel.X, 0, vel.Z).Magnitude
		local usingStairs = isUsingStairs(character, humanoid, rootPart)
		local onGround = (humanoid.FloorMaterial ~= Enum.Material.Air) or usingStairs

		if onGround then
			-- Se stava volando in modo sospetto e tocca terra con spinta/velocità anomala, lo riporta indietro
			if antiFlingEnabled and not isJumping and inSuspiciousAir and (speed >= 32 or vel.Y < -18) then
				teleportToSafeGround(rootPart, humanoid)
				return
			end

			fallTime = 0
			inSuspiciousAir = false
			if jumpLockTimer <= 0 then
				isJumping = false
			end

			local now = os.clock()
			if speed < 35 and (now - lastSaveTime) >= 0.25 then
				previousSafeGroundCFrame = safeGroundCFrame
				safeGroundCFrame = rootPart.CFrame
				lastSaveTime = now
			end
			return
		end

		if not antiFlingEnabled or isJumping then
			fallTime = 0
			inSuspiciousAir = false
			return
		end

		-- Non scatta quando sali normalmente (vel.Y > 1 e niente fling orizzontale)
		if vel.Y > 1 and horizontalSpeed < 45 then
			fallTime = 0
			return
		end

		if vel.Y < -2 or horizontalSpeed >= 45 then
			inSuspiciousAir = true
			fallTime = fallTime + dt
			if fallTime >= FALL_THRESHOLD then
				teleportToSafeGround(rootPart, humanoid)
			end
		else
			fallTime = 0
		end
	end)
end
