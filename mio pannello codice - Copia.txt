local function makeDraggable(frame, handle)
    local dragging, dragInput, dragStart, startPos
    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local function createToggle(name, labelText, callback)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(1, 0, 0, 40)
    btn.BackgroundColor3 = COLOR_OFF
    btn.Text = labelText .. ": OFF"
    btn.TextColor3 = TEXT_COLOR
    btn.Font = Enum.Font.SourceSansBold
    btn.TextSize = 16
    btn.Parent = Container

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn

    btn.MouseButton1Click:Connect(function()
        states[name] = not states[name]
        if states[name] then
            btn.BackgroundColor3 = COLOR_ON
            btn.Text = labelText .. ": ON"
        else
            btn.BackgroundColor3 = COLOR_OFF
            btn.Text = labelText .. ": OFF"
        end
        callback(states[name], btn)
    end)
    return btn
end

local flyBV = nil
local flyBG = nil
local animConnection = nil

createToggle("Fly", "Fly Player", function(active)
    local character = LocalPlayer.Character
    if not character or not character:FindFirstChild("HumanoidRootPart") then return end
    local hrp = character.HumanoidRootPart
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local animateScript = character:FindFirstChild("Animate")

    flySpeedFrame.Visible = active

    if active then
        if flyBV then flyBV:Destroy() end
        if flyBG then flyBG:Destroy() end
        if animConnection then animConnection:Disconnect() end

        if animateScript then animateScript.Enabled = false end

        for _, track in pairs(humanoid:GetPlayingAnimationTracks()) do
            track:Stop(0)
        end

        animConnection = humanoid.AnimationPlayed:Connect(function(track)
            track:Stop(0)
        end)

        humanoid:ChangeState(Enum.HumanoidStateType.Swimming)

        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(1, 1, 1) * math.huge
        flyBG.P = 90000
        flyBG.CFrame = hrp.CFrame
        flyBG.Parent = hrp

        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(1, 1, 1) * math.huge
        flyBV.Velocity = Vector3.new(0, 0, 0)
        flyBV.Parent = hrp

        task.spawn(function()
            while states.Fly and character and hrp and humanoid do
                local cam = workspace.CurrentCamera
                local moveDirection = humanoid.MoveDirection
                
                humanoid:ChangeState(Enum.HumanoidStateType.Swimming)

                local lookAt = cam.CFrame.LookVector
                flyBG.CFrame = CFrame.new(hrp.Position, hrp.Position + Vector3.new(lookAt.X, lookAt.Y, lookAt.Z))

                if moveDirection.Magnitude > 0 then
                    local finalVelocity = moveDirection * forcedFlySpeed
                    if math.abs(lookAt.Y) > 0.2 then
                        finalVelocity = Vector3.new(finalVelocity.X, lookAt.Y * forcedFlySpeed * (moveDirection.Z < 0 and -1 or (moveDirection.Z > 0 and 1 or 0)), finalVelocity.Z)
                    end
                    flyBV.Velocity = finalVelocity
                else
                    flyBV.Velocity = Vector3.new(0, 0, 0)
                end
                task.wait()
            end
            
            if flyBV then flyBV:Destroy() end
            if flyBG then flyBG:Destroy() end
            if animConnection then animConnection:Disconnect() end
            if animateScript then animateScript.Enabled = true end
            if humanoid then humanoid:ChangeState(Enum.HumanoidStateType.Standing) end
        end)
    else
        if flyBV then flyBV:Destroy() end
        if flyBG then flyBG:Destroy() end
        if animConnection then animConnection:Disconnect() end
        if animateScript then animateScript.Enabled = true end
        if humanoid then humanoid:ChangeState(Enum.HumanoidStateType.Standing) end
    end
end)

RunService.RenderStepped:Connect(function()
    if states.Aimbot then
        local closest = nil
        local dist = math.huge
        local cam = workspace.CurrentCamera
        local myPos = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart") and LocalPlayer.Character.HumanoidRootPart.Position
        
        if myPos then
            for _, p in pairs(Players:GetPlayers()) do
                if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                    local d = (p.Character.HumanoidRootPart.Position - myPos).Magnitude
                    if d < dist then
                        dist = d
                        closest = p.Character.HumanoidRootPart
                    end
                end
            end
            if closest then
                cam.CFrame = CFrame.new(cam.CFrame.Position, closest.Position)
            end
        end
    end
end)

RunService.Stepped:Connect(function()
    local character = LocalPlayer.Character
    if character then
        if states.Noclip then
            for _, part in pairs(character:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
        
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            if not states.Fly then
                humanoid.WalkSpeed = forcedSpeed
            end
            
            if states.AntiRagdoll then
                humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
                humanoid.PlatformStand = false
            end
        end
        
        if states.Freeze and character:FindFirstChild("HumanoidRootPart") then
            character.HumanoidRootPart.Anchored = true
        end
    end
end)

        params.FilterDescendantsInstances = {LocalPlayer.Character}
        params.FilterType = Enum.RaycastFilterType.Exclude
        local result = workspace:Raycast(hrp.Position, hrp.CFrame.LookVector * 5, params)
        if result then
            hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, 50, hrp.AssemblyLinearVelocity.Z)
        end
    end

    if states.FlyJumpHappy then
        for i = 1, 5 do 
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
            task.wait(0.01)
            hum:ChangeState(Enum.HumanoidStateType.Swimming)
        end
    elseif states.FlyJump then
        hum:ChangeState(Enum.HumanoidStateType.Jumping)
    end
end)

createToggle("OldRoblox", "Old Roblox Mode", function(active)
    if active then
        buildOldUI()
        pcall(function()
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Chat, false)
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.PlayerList, false)
        end)

        task.spawn(function()
            while states.OldRoblox do
                local character = LocalPlayer.Character
                if character then
                    for _, joint in pairs(character:GetDescendants()) do
                        if joint:IsA("Motor6D") and (
                            joint.Name:find("Elbow") or joint.Name:find("Knee") or 
                            joint.Name:find("Wrist") or joint.Name:find("Ankle")
                        ) then
                            joint.C0 = CFrame.new(joint.C0.Position)
                            joint.C1 = CFrame.new(joint.C1.Position)
                        end
                    end
                    local animateScript = character:FindFirstChild("Animate")
                    if animateScript and animateScript:FindFirstChild("idle") then
                        animateScript.idle.Animation1.AnimationId = "rbxassetid://180426354"
                    end
                end
                task.wait(0.1)
            end
        end)
    else
        if NostalgiaGui then 
            NostalgiaGui:Destroy() 
            NostalgiaGui = nil
        end
        pcall(function()
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.Chat, true)
            StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.PlayerList, true)
        end)

        local character = LocalPlayer.Character
        local animateScript = character and character:FindFirstChild("Animate")
        if animateScript then
            animateScript.Enabled = false
            task.wait(0.1)
            animateScript.Enabled = true
        end
    end
end)
