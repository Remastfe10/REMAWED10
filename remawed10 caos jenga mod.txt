finishButton.BackgroundColor3 = Color3.fromRGB(200, 50, 50)

-- Arrotondamento angoli Pulsante
local buttonCorner = Instance.new("UICorner")
buttonCorner.CornerRadius = UDim.new(0, 6)
buttonCorner.Parent = finishButton

-- Funzione di Teletrasporto alla Part "Button"
local function finishTheGame()
	local character = localPlayer.Character
	if not character then return end
	
	local hrp = character:FindFirstChild("HumanoidRootPart")
	if not hrp then return end

	-- Cerca la Part chiamata "Button" nel Workspace
	local targetPart = workspace:FindFirstChild("Button", true)

	if targetPart and targetPart:IsA("BasePart") then
		-- Teletrasporta il giocatore sopra la part "Button"
		hrp.CFrame = targetPart.CFrame + Vector3.new(0, 3, 0)
	else
		warn("La Part 'Button' non è stata trovata nel Workspace!")
	end
end

-- Collegamento del click al pulsante
finishButton.MouseButton1Click:Connect(finishTheGame)
